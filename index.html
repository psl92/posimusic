export default { async fetch(request, env) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET,OPTIONS", "Access-Control-Allow-Headers": "*", "Access-Control-Max-Age": "86400" }; if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders }); const url = new URL(request.url); const bucket = env.busimusic; const path = url.pathname; // 获取全部资源列表接口 if (path === "/getResourceList") { try { const listData = await buildResourceList(bucket); return Response.json(listData, { headers: corsHeaders }); } catch (e) { return Response.json({ error: e.message }, { status: 500, headers: corsHeaders }); } } // 获取歌词文件接口 if (path.startsWith("/getLyric")) { const lyricName = url.searchParams.get("name"); if (!lyricName) return Response.json({ msg: "缺少参数name" }, { status:400, headers:corsHeaders }); const lyricObj = await bucket.get(`lyric/${lyricName}`); if (!lyricObj) return Response.json({ msg:"歌词不存在" },{status:404,headers:corsHeaders}); const lyricText = await lyricObj.text(); return new Response(lyricText, { headers:{...corsHeaders,"Content-Type":"text/plain;charset=utf-8"} }); } // 静态资源分发 const routeMap = { "/music/": "music/", "/img/": "img/", "/video/": "video/", "/reward/": "reward/" }; for(let prefix in routeMap){ if(path.startsWith(prefix)){ const fileKey = routeMap[prefix] + decodeURIComponent(path.slice(prefix.length)); const fileObj = await bucket.get(fileKey); if(!fileObj) return Response.json({msg:"文件不存在"},{status:404,headers:corsHeaders}); const headers = new Headers(corsHeaders); if(fileKey.endsWith(".mp3")) headers.set("Content-Type","audio/mpeg"); if(fileKey.endsWith(".mp4")) headers.set("Content-Type","video/mp4"); if(fileKey.match(/\.(jpg|jpeg|png|webp)$/)) headers.set("Content-Type","image/*"); return new Response(fileObj.body, { headers }); } } return Response.json({msg:"接口不存在"},{status:404,headers:corsHeaders}); } }; // 分页遍历R2全部文件 async function listAll(bucket, prefix) { let list = []; let cursor = null; do { const res = await bucket.list({ prefix, cursor }); list = list.concat(res.objects); cursor = res.cursor; } while (cursor); return list; } // 构建完整资源数组（自动拆分歌手 - 歌名） async function buildResourceList(bucket){ const out = []; let idCounter = 1; async function safeList(prefix){ try{ return await listAll(bucket,prefix); }catch(e){ return []; } } const musicObjs = await safeList("music/"); const imgObjs = await safeList("img/"); const lyricObjs = await safeList("lyric/"); const videoObjs = await safeList("video/"); const rewardObjs = await safeList("reward/"); function getStem(filename){ const dot = filename.lastIndexOf("."); return dot>0 ? filename.substring(0,dot) : filename; } // 音乐资源处理 for(const o of musicObjs){ const fullName = o.key.slice("music/"); const stem = getStem(fullName); let coverImg = null; let lyricName = null; let singer = "未知歌手"; let songName = stem; if(stem.includes(" - ")){ const splitArr = stem.split(" - "); singer = splitArr[0].trim(); songName = splitArr[1].trim(); } // 匹配封面图 for(const img of imgObjs){ const imgName = img.key.slice("img/"); const imgStem = getStem(imgName); if(imgStem === stem){ coverImg = `https://api.busisi.com/img/${encodeURIComponent(imgName)}`; break; } } // 匹配歌词 for(const ly of lyricObjs){ const lyName = ly.key.slice("lyric/"); const lyStem = getStem(lyName); if(lyStem === stem){ lyricName = lyName; break; } } out.push({ id:idCounter++, type:"music", name:songName, singer:singer, audioLink:`https://api.busisi.com/music/${encodeURIComponent(fullName)}`, coverImg:coverImg, lyricName:lyricName }) } // 视频资源处理 for(const o of videoObjs){ const fullName = o.key.slice("video/"); const stem = getStem(fullName); let coverImg = null; let singer = "未知歌手"; let videoName = stem; if(stem.includes(" - ")){ const splitArr = stem.split(" - "); singer = splitArr[0].trim(); videoName = splitArr[1].trim(); } for(const img of imgObjs){ const imgName = img.key.slice("img/"); const imgStem = getStem(imgName); if(imgStem === stem){ coverImg = `https://api.busisi.com/img/${encodeURIComponent(imgName)}`; break; } } out.push({ id:idCounter++, type:"video", name:videoName, singer:singer, videoLink:`https://api.busisi.com/video/${encodeURIComponent(fullName)}`, coverImg:coverImg, lyricName:null }) } // 图片资源 for(const o of imgObjs){ const fullName = o.key.slice("img/"); const stem = getStem(fullName); out.push({ id:idCounter++, type:"image", name:stem, singer:"", coverImg:`https://api.busisi.com/img/${encodeURIComponent(fullName)}` }) } // 赞赏图片资源 for(const o of rewardObjs){ const fullName = o.key.slice("reward/"); const stem = getStem(fullName); out.push({ id:idCounter++, type:"reward", name:stem, singer:"", coverImg:`https://api.busisi.com/reward/${encodeURIComponent(fullName)}` }) } return out; }
